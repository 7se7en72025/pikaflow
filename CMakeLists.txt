cmake_minimum_required(VERSION 3.24)
project(PikaFlow VERSION 0.4.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PIKAFLOW_ENABLE_LLVM "Build LLVM native and JIT backends" ON)
option(PIKAFLOW_BUILD_LLVM_PLUGIN "Build the standalone LLVM pass plugin when supported" ON)
include(CTest)
find_package(Threads REQUIRED)

add_library(pikaflow_runtime STATIC src/runtime.cpp)
target_include_directories(pikaflow_runtime PUBLIC include)
target_link_libraries(pikaflow_runtime PUBLIC Threads::Threads)
set_target_properties(pikaflow_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(pikaflow_core STATIC src/common.cpp src/lexer.cpp src/parser.cpp
    src/compiler.cpp src/ir.cpp src/interpreter.cpp src/optimizer.cpp)
target_include_directories(pikaflow_core PUBLIC include)
target_link_libraries(pikaflow_core PUBLIC pikaflow_runtime)

add_executable(pikaflow src/main.cpp)
target_link_libraries(pikaflow PRIVATE pikaflow_core)
target_compile_definitions(pikaflow PRIVATE
    PF_VERSION="${PROJECT_VERSION}"
    PF_RUNTIME_LIBRARY="$<TARGET_FILE:pikaflow_runtime>"
    PF_COMPILER_DRIVER="${CMAKE_CXX_COMPILER}"
    PF_STDLIB_DIR="${CMAKE_CURRENT_SOURCE_DIR}/stdlib")

if(PIKAFLOW_ENABLE_LLVM)
    find_package(LLVM REQUIRED CONFIG)
    message(STATUS "PikaFlow LLVM ${LLVM_PACKAGE_VERSION}: ${LLVM_DIR}")
    if(NOT LLVM_VERSION_MAJOR EQUAL 24)
        message(FATAL_ERROR "This PikaFlow revision targets LLVM 24; select the documented LLVM revision")
    endif()
    llvm_map_components_to_libnames(PIKAFLOW_LLVM_LIBRARIES
        core support native nativecodegen passes orcjit executionengine
        runtimedyld jitlink target mc codegen scalaropts instcombine)
    if(NOT WIN32)
        llvm_map_components_to_libnames(PIKAFLOW_ABI_LIBRARY abi)
        list(APPEND PIKAFLOW_LLVM_LIBRARIES ${PIKAFLOW_ABI_LIBRARY})
    endif()
    add_library(pikaflow_backend STATIC src/backend.cpp src/llvm/CheckedArithmeticPass.cpp)
    if(MSVC AND NOT "$ENV{LIB}" STREQUAL "")
        set(PIKAFLOW_NATIVE_LIBRARY_DIRS "$ENV{LIB}" CACHE STRING "SDK and CRT library directories for native linking" FORCE)
    endif()
    set(PF_MSVC_LIBRARY_DIRS "${PIKAFLOW_NATIVE_LIBRARY_DIRS}")
    string(REPLACE "\\" "\\\\" PF_MSVC_LIBRARY_DIRS "${PF_MSVC_LIBRARY_DIRS}")
    string(REPLACE "\"" "\\\"" PF_MSVC_LIBRARY_DIRS "${PF_MSVC_LIBRARY_DIRS}")
    string(REPLACE "\n" "\\n" PF_MSVC_LIBRARY_DIRS "${PF_MSVC_LIBRARY_DIRS}")
    string(REPLACE "\r" "\\r" PF_MSVC_LIBRARY_DIRS "${PF_MSVC_LIBRARY_DIRS}")
    configure_file(cmake/BackendConfig.hpp.in generated/pikaflow/backend_config.hpp @ONLY)
    target_include_directories(pikaflow_backend PUBLIC include)
    target_include_directories(pikaflow_backend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_include_directories(pikaflow_backend SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
    separate_arguments(PIKAFLOW_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
    target_compile_options(pikaflow_backend PRIVATE ${PIKAFLOW_LLVM_DEFINITIONS})
    if(MSVC)
        target_compile_options(pikaflow_backend PRIVATE /utf-8 /wd4624)
    endif()
    target_link_libraries(pikaflow_backend PUBLIC pikaflow_core ${PIKAFLOW_LLVM_LIBRARIES})
    target_link_libraries(pikaflow PRIVATE pikaflow_backend)
    target_compile_definitions(pikaflow PRIVATE PF_HAS_LLVM=1)
    if(PIKAFLOW_BUILD_LLVM_PLUGIN AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND
       LLVM_ENABLE_PLUGINS AND LLVM_LINK_LLVM_DYLIB AND TARGET LLVM)
        list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
        include(AddLLVM)
        add_llvm_pass_plugin(pikaflow_checked_passes
            src/llvm/PassPlugin.cpp src/llvm/CheckedArithmeticPass.cpp)
        set_target_properties(pikaflow_checked_passes PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
        target_include_directories(pikaflow_checked_passes SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
        target_compile_options(pikaflow_checked_passes PRIVATE ${PIKAFLOW_LLVM_DEFINITIONS})
        target_compile_definitions(pikaflow_checked_passes PRIVATE PF_PLUGIN_VERSION="${PROJECT_VERSION}")
        install(TARGETS pikaflow_checked_passes LIBRARY DESTINATION lib/pikaflow/plugins)
    elseif(PIKAFLOW_BUILD_LLVM_PLUGIN)
        message(STATUS "Standalone LLVM pass plugin unavailable in this LLVM configuration; compiler pass remains enabled")
    endif()
endif()

if(MSVC)
    target_compile_options(pikaflow_core PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(pikaflow_runtime PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(pikaflow PRIVATE /W4 /permissive- /utf-8)
else()
    foreach(target pikaflow_core pikaflow_runtime pikaflow)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()
